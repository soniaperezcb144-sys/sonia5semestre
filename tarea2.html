CIRCUITO 01

Detector de humo y gas (MQ‑2)

Sensor de gas MQ‑2, tres LEDs de estado (verde / amarillo / rojo) y un buzzer de alarma según el nivel de humo detectado.

MQ‑2 3× LED buzzer
detector_humo.ino 
// C++ code
int humo;
int rojo=11;
int amarillo=12;
int verde=13;
int alarma =10;

void setup()
{
  Serial.begin(9600); //usar monitor en serie
  pinMode(rojo,OUTPUT);// salidas led
  pinMode(amarillo,OUTPUT);
  pinMode(verde,OUTPUT);
  pinMode(alarma,OUTPUT);//salida alarma
}

void loop()
{
  humo = analogRead(A5);

  if (humo<930)
  {
    digitalWrite(verde,HIGH);
    digitalWrite(amarillo,LOW);
    digitalWrite(rojo,LOW);
    noTone(alarma);
    Serial.println("ESTAS FUERA DE PELIGRO");
    delay(500);
  }

  if ((humo>930) and (humo<960))
  {
    digitalWrite(verde,LOW);
    digitalWrite(amarillo,HIGH);
    digitalWrite(rojo,LOW);
    noTone(alarma);
    Serial.println(" OJO! EL PELIGRO SE ACERCA");
    delay(500);
  }

  if (humo>960)
  {
    digitalWrite(verde,LOW);
    digitalWrite(amarillo,LOW);
    digitalWrite(rojo,HIGH);
    tone(alarma,440);
    Serial.println("ESTAS EN PELIGRO");
    delay(500);
  }
}

CIRCUITO 03

Sensor de temperatura (TMP36)

Lectura de temperatura con sensor TMP36 en A0; al superar 50 °C se enciende un LED y se activa un zumbador de aviso.

TMP36 LED buzzer
temperatura.ino 
// C++ code
/*
  Realizado por Luis Manuel Giraldo Arias
  Profesor David Martínez
  Curso Arduino en Arte Mecatrónica
  Junio 1 de 2020
  Lectura de sensor de temperatura, led que
  enciende y zumbador
*/
void setup()
{
  pinMode(A0, INPUT);
  pinMode(8, OUTPUT);
  pinMode(13, OUTPUT);
  Serial.begin(9600);
}

void loop()
{
  if ((-40 + 0.488155 * (analogRead(A0) - 20)) >= 50) {
    //  Se lee sensor de temperatura y se compara con 50
    // Si es mayor o igual a 50 enciende led y se activa zumbador
    digitalWrite(8, HIGH);
    digitalWrite(13, HIGH);
  } else {
    // Si lectura de sensor es menor de 50 no se enciende led ni zumbador
    digitalWrite(8, LOW);
    digitalWrite(13, LOW);
  }
  delay(300); // Wait for 300 millisecond(s)
  Serial.println((-40 + 0.488155 * (analogRead(A0) - 20)));
}
